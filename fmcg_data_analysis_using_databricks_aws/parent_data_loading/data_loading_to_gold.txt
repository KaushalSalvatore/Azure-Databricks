COPY INTO fmcg.fmcg_gold.fact_orders
FROM (
  SELECT 
    CAST(date AS DATE) AS date,
    product_code,
    customer_code,
    CAST(sold_quantity AS BIGINT) AS sold_quantity
  FROM '/Volumes/fmcg/fmcg_gold/parent_incremental_update/fact_orders.csv'
)

FORMAT_OPTIONS ('header' = 'true');
